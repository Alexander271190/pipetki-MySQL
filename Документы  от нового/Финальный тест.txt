        const updatedHtml = modal.innerHTML;
        const hasIcons = updatedHtml.includes('fa-check') || updatedHtml.includes('fa-xmark');
        log('UI-PW-ICONS', 'иконки валидации (fa-check/xmark)', hasIcons ? 'PASS' : 'FAIL');
      } else {
        log('UI-PW-ICONS', 'поле ввода пароля', 'FAIL', 'не найдено');
      }
    } else {
      log('UI-PW-POLICY', 'модалка открыта', 'FAIL');
    }
    uiStore.close();
  }

  /* ═════════════════════════════════════════════════════════════
     БЛОК 48. UI — UserEditModal (селект pageSizes)
     ═════════════════════════════════════════════════════════════ */
  console.log('\n%c📏 48. UI — UserEditModal размеры', 'font-size:14px;font-weight:bold;color:#16a34a;');
  {
    uiStore.open('userEdit', null);
    await sleep(600);
    const modal = document.querySelector('.modal-overlay.active .modal');
    if (modal) {
      const selects = Array.from(modal.querySelectorAll('select'));
      const pageSizeSelect = selects.find(s => {
        const opts = Array.from(s.querySelectorAll('option')).map(o => o.value);
        return opts.includes('50') || opts.includes('100');
      });
      if (pageSizeSelect) {
        const opts = Array.from(pageSizeSelect.querySelectorAll('option'))
          .map(o => o.value).filter(v => v && !isNaN(parseInt(v, 10)));
        const expectedSizes = settingsStore.getTableViewSettings?.().pageSizes?.map(String) || ['50','100','200','500'];
        const match = expectedSizes.every(s => opts.includes(s));
        log('UI-PAGE-SIZES', 'селект использует pageSizes',
            match ? 'PASS' : 'FAIL',
            `опции: [${opts.join(',')}] ожидалось: [${expectedSizes.join(',')}]`);
      } else {
        log('UI-PAGE-SIZES', 'селект «Строк на странице»', 'FAIL', 'не найден');
      }
    } else {
      log('UI-PAGE-SIZES', 'модалка открыта', 'FAIL');
    }
    uiStore.close();
  }

  /* ═════════════════════════════════════════════════════════════
     БЛОК 49. Modal settings resize (наш фикс)
     ═════════════════════════════════════════════════════════════ */
  console.log('\n%c🪟 49. Modal settings resize', 'font-size:14px;font-weight:bold;color:#16a34a;');
  {
    uiStore.open('settings');
    await sleep(400);
    const modal = document.querySelector('.modal-overlay.active .modal');
    if (modal) {
      const hasClass = modal.classList.contains('modal-settings-resizable');
      log('UI', 'modal-settings-resizable class', hasClass ? 'PASS' : 'FAIL');
      if (hasClass) {
        const cs = getComputedStyle(modal);
        log('UI', 'resize=both', cs.resize === 'both' ? 'PASS' : 'FAIL', cs.resize);
        log('UI', 'overflow=auto/hidden', ['auto', 'hidden'].includes(cs.overflow?.split(' ')[0]) ? 'PASS' : 'FAIL', cs.overflow);
      }
    } else {
      log('UI', 'модалка настроек открыта', 'FAIL');
    }
    uiStore.close();
    await sleep(200);
  }

  /* ═════════════════════════════════════════════════════════════
     БЛОК 50. Логин админом (наш фикс)
     ═════════════════════════════════════════════════════════════ */
  console.log('\n%c🔤 50. Логин админом', 'font-size:14px;font-weight:bold;color:#16a34a;');
  {
    const testLogin = 'v8_rename_' + uid;
    const cr = await api('/users', 'POST', {
      login: testLogin, fullName: 'V8 Rename', position: 'Test',
      role: 'user', onlyOwnDepartment: false, extraPermissions: [], isActing: false
    });
    if (!cr.ok) {
      log('LOGIN', 'создан тестовый', 'FAIL', cr.body?.error);
    } else {
      created.users.push({ id: cr.body.id, login: testLogin });
      log('LOGIN', 'создан тестовый', 'PASS');
      const newLogin = testLogin + '_new';
      const put = await api(`/users/${cr.body.id}`, 'PUT', {
        login: newLogin, fullName: 'V8 Rename', position: 'Test',
        role: 'user', onlyOwnDepartment: false, extraPermissions: [], isActing: false
      });
      log('LOGIN', 'rename админом', put.ok ? 'PASS' : 'FAIL', put.body?.error || '');
      const check = (await api('/users')).body?.find(x => x.id === cr.body.id);
      log('LOGIN', 'логин фактически изменён', check?.login === newLogin ? 'PASS' : 'FAIL', check?.login);
      const dup = await api(`/users/${cr.body.id}`, 'PUT', {
        login: 'admin', fullName: 'V8 Rename', position: 'Test',
        role: 'user', onlyOwnDepartment: false, extraPermissions: [], isActing: false
      });
      log('LOGIN', 'занять admin → 409/403', [409, 403].includes(dup.status) ? 'PASS' : 'FAIL', `status=${dup.status}`);
    }
    const userLogin = 'v8_norename_' + uid;
    const cr2 = await api('/users', 'POST', {
      login: userLogin, fullName: 'V8 NoRename', position: 'Test',
      role: 'user', onlyOwnDepartment: false, extraPermissions: ['manage_users'], isActing: false
    });
    if (cr2.ok) {
      created.users.push({ id: cr2.body.id, login: userLogin });
      const loginR = await api('/auth/login', 'POST', { login: userLogin, password: cr2.body.tempPassword }, null);
      if (loginR.ok) {
        const tmpTok = loginR.body.token;
        await api('/auth/change-password', 'POST', {
          currentPassword: cr2.body.tempPassword,
          newPassword: 'V8Pass1!', confirmPassword: 'V8Pass1!'
        }, tmpTok);
        const reLogin = await api('/auth/login', 'POST', { login: userLogin, password: 'V8Pass1!' }, null);
        if (reLogin.ok) {
          const userTok = reLogin.body.token;
          const put2 = await api(`/users/${cr2.body.id}`, 'PUT', {
            login: userLogin + '_hack', fullName: 'V8 NoRename', position: 'Test',
            role: 'user', onlyOwnDepartment: false, extraPermissions: ['manage_users'], isActing: false
          }, userTok);
          log('LOGIN', 'не-админ не может сменить логин → 403', put2.status === 403 ? 'PASS' : 'FAIL', `status=${put2.status}`);
        }
      }
    }
  }

  /* ═════════════════════════════════════════════════════════════
     БЛОК 51. Barcode max length в авто-ШК (наш фикс)
     ═════════════════════════════════════════════════════════════ */
  console.log('\n%c🏷 51. barcode_max_length в авто-ШК', 'font-size:14px;font-weight:bold;color:#16a34a;');
  {
    const origBcl = origSettings.barcode_max_length || '255';
    await api('/settings/system', 'PUT', { barcode_max_length: '20' });
    await sleep(300);
    try { await fetch('/api/settings/system', { headers: { 'Authorization': 'Bearer ' + adminToken } }); } catch {}
    const testId = `V8-BC-${uid}`;
    const longSerial = 'S' + 'X'.repeat(60);
    const r = await api('/pipettes', 'POST', {
      id: testId, model: 'V8 BarcodeTest', equipmentType: 'pipette',
      lastCalibration: today, responsible: 'Admin', serial: longSerial
    });
    if (r.ok) {
      created.pipettes.push(testId);
      const fresh = await api(`/pipettes/${encodeURIComponent(testId)}`);
      const bc = String(fresh.body?.barcode || '');
      log('BARCODE-LIMIT', 'barcode ≤ 20 символов', bc.length > 0 && bc.length <= 20 ? 'PASS' : 'FAIL', `len=${bc.length}`);
    } else {
      log('BARCODE-LIMIT', 'создание прибора', 'FAIL', r.body?.error);
    }
    await api('/settings/system', 'PUT', { barcode_max_length: origBcl });
    await sleep(200);
  }

  } finally {

  /* ═════════════════════════════════════════════════════════════
     CLEANUP
     ═════════════════════════════════════════════════════════════ */
  console.log('\n%c🧹 CLEANUP', 'font-size:14px;font-weight:bold;color:#4f46e5;');

  // Восстановить системные настройки
  if (origSettings && Object.keys(origSettings).length > 0) {
    try {
      const restore = {};
      const keys = [
        'max_batch_size', 'max_bulk_size', 'max_model_length', 'max_serial_length',
        'password_min_length', 'password_max_length',
        'password_require_lower', 'password_require_upper',
        'password_require_digit', 'password_require_special', 'password_forbid_spaces',
        'reminders_enabled', 'reminders_for_roles',
        'impersonation_mode',
        'page_sizes', 'default_page_size', 'table_density',
        'barcode_max_length',
      ];
      for (const k of keys) {
        if (origSettings[k] !== undefined) restore[k] = origSettings[k];
      }
      if (Object.keys(restore).length > 0) {
        await api('/settings/system', 'PUT', restore);
        await settingsStore.loadAll();
        console.log('✅ Настройки системы восстановлены');
      }
    } catch (e) {
      console.warn('[MEGA-TEST] restore settings failed:', e);
    }
  }

  // Восстановить export/filters
  if (_origExportCfg) { try { await api('/settings/export', 'PUT', _origExportCfg); } catch {} }
  if (_origFiltersCfg) { try { await api('/settings/filters', 'PUT', _origFiltersCfg); } catch {} }

  // Восстановить effective print
  try {
    const e = await api('/settings/print-documents/effective');
    if (e.ok) settingsStore.printDocs = e.body;
  } catch {}

  // Восстановить window.open
  restoreOpen();

  await sleep(500);

  // Удалить созданные сущности
  for (const u of created.users) {
    try { await api('/users/' + u.id, 'DELETE'); } catch {}
  }
  for (const id of created.presets) {
    try { await api('/users/presets/' + id, 'DELETE'); } catch {}
  }
  for (const id of [...created.pipettes, ...created.lifecycle]) {
    try {
      await api(`/pipettes/${encodeURIComponent(id)}/lock?force=true`, 'DELETE').catch(() => {});
      await api(`/pipettes/${encodeURIComponent(id)}`, 'DELETE');
    } catch {}
  }
  for (const id of created.roles) {
    try { await api('/roles/' + id, 'DELETE'); } catch {}
  }
  for (const id of created.printers) {
    try { await api('/barcode-printers/' + id, 'DELETE'); } catch {}
  }
  for (const id of created.eqTpls) {
    try { await api('/equipment-templates/' + id, 'DELETE'); } catch {}
  }
  for (const id of created.viewTpls) {
    try { await api('/view-templates/' + id, 'DELETE'); } catch {}
  }

  // Обновить UI
  try { await pipStore.loadPaged(); } catch {}
  try { await statsStore.load(); } catch {}
  try { await settingsStore.loadAll(); } catch {}

  window.__megaTestV80Running = false;

  /* ═════════════════════════════════════════════════════════════
     ИТОГ
     ═════════════════════════════════════════════════════════════ */
  const p = results.filter(r => r.s === 'PASS').length;
  const f = results.filter(r => r.s === 'FAIL').length;
  const sk = results.filter(r => r.s === 'SKIP').length;
  const elapsed = ((Date.now() - T0) / 1000).toFixed(1);

  console.log('\n' + '═'.repeat(60));
  console.log(`%c📊 v8.0 ИТОГ: ✅ ${p}  ❌ ${f}  ⏭️ ${sk}  (всего: ${results.length}) · ${elapsed}s`,
    `font-size:18px;font-weight:bold;color:${f ? '#dc2626' : '#16a34a'};`);
  console.log('═'.repeat(60));

  // Группировка по блокам
  if (f > 0) {
    console.log('%c❌ ПРОВАЛЫ (по блокам):', 'color:#dc2626;font-weight:bold;');
    const byGrp = {};
    results.filter(r => r.s === 'FAIL').forEach(r => {
      if (!byGrp[r.grp]) byGrp[r.grp] = [];
      byGrp[r.grp].push(r);
    });
    for (const [grp, items] of Object.entries(byGrp)) {
      console.group(`[${grp}] — ${items.length} провал(ов)`);
      items.forEach(r => console.log(`  ❌ ${r.name}${r.d ? ' — ' + r.d : ''}`));
      console.groupEnd();
    }
  }

  if (sk > 0) {
    console.log('%c⏭️ SKIP:', 'color:#f59e0b;font-weight:bold;');
    console.table(results.filter(r => r.s === 'SKIP'));
  }

  if (f === 0) {
    console.log('%c🎉 v8.0 — всё зелёное! Проект готов.', 'color:#16a34a;font-weight:bold;font-size:16px;');
  }

  // Сохранить отчёт
  try {
    sessionStorage.setItem('pipette_mega_v80_results', JSON.stringify({
      ts: Date.now(), elapsed: Number(elapsed),
      pass: p, fail: f, skip: sk, total: results.length,
      results
    }));
  } catch {}
  window.__megaTestV80Results = results;

  // Кнопка скачивания
  window.__downloadV80 = () => {
    const blob = new Blob([JSON.stringify({
      ts: new Date().toISOString(), elapsed: Number(elapsed),
      pass: p, fail: f, skip: sk, total: results.length,
      results
    }, null, 2)], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'mega-test-v80-' + new Date().toISOString().slice(0, 19).replace(/[T:]/g, '-') + '.json';
    a.click();
    URL.revokeObjectURL(a.href);
  };

  if (f > 0 || sk > 0) {
    console.log('%c💾 Скачать JSON-отчёт: window.__downloadV80()', 'color:#8b5cf6;font-weight:bold');
  }

  } // finally
})();
