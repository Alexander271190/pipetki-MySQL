3.3. Удали sanitizeCsvCell() — она больше нигде не используется
js
function sanitizeCsvCell(value) {
  let s = String(value == null ? '' : value);
  if (/^[=+\-@\t\r\n|]/.test(s)) {
    s = "'" + s;
  }
  s = s.replace(/"/g, '""');
  return /[";\n\r]/.test(s) ? '"' + s + '"' : s;
}
