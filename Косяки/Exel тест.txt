📊 Excel-файл для теста импорта (500–1000 записей)
Файл генерируется через exceljs (уже в зависимостях проекта). Сохрани как gen-import-xlsx.js в корне проекта и запусти:

bash
node gen-import-xlsx.js 1000
Создастся файл test-import-1000.xlsx — настоящий Excel-формат (не CSV), с шапкой, автошириной, зеброй, выравниванием.

function randomDate() {
  const start = new Date();
  start.setFullYear(start.getFullYear() - 3);
  const end = new Date();
  const d = new Date(start.getTime() + Math.random() * (end.getTime() - start.getTime()));
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');

  // 30% ISO, 35% DD.MM.YYYY, 35% MM.DD.YYYY
  const r = Math.random();
  if (r < 0.30) return `${y}-${m}-${day}`;
  if (r < 0.65) return `${day}.${m}.${y}`;
  return `${m}.${day}.${y}`;
}

async function generate() {
  const wb = new ExcelJS.Workbook();
  wb.creator = 'Pipette App Test';
  wb.created = new Date();

  const ws = wb.addWorksheet('Оборудование', {
    views: [{ state: 'frozen', ySplit: 1 }],
  });

    { header: 'Серийный номер',        key: 'serial',     width: 16 },
    { header: 'Производитель',         key: 'manufacturer', width: 16 },
    { header: 'Модель',                key: 'model',      width: 20 },
    { header: 'Тип оборудования',      key: 'equipmentType', width: 18 },
    { header: 'Объём (мкл)',           key: 'volume',     width: 12 },
    { header: 'Отдел',                 key: 'department', width: 24 },
    { header: 'МПИ',                   key: 'interval',   width: 8 },
    { header: 'Дата поверки',          key: 'lastCalibration', width: 14 },
    { header: 'Свидетельство',         key: 'cert',       width: 18 },
    { header: 'Результат',             key: 'result',     width: 14 },
    { header: 'Статус',                key: 'active',     width: 10 },
    { header: 'Ответственный',         key: 'responsible', width: 18 },
    { header: 'Место хранения',        key: 'location',   width: 16 },
    { header: 'Примечание',            key: 'notes',      width: 24 },
  ];

  // Стиль шапки
  const headerRow = ws.getRow(1);
  headerRow.height = 24;
  headerRow.eachCell(cell => {
    cell.font      = { bold: true, color: { argb: 'FFFFFFFF' }, size: 11 };
    cell.fill      = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF1E293B' } };
    cell.alignment = { vertical: 'middle', horizontal: 'left' };
    cell.border    = {
      top:    { style: 'thin', color: { argb: 'FFCBD5E1' } },
      left:   { style: 'thin', color: { argb: 'FFCBD5E1' } },
      bottom: { style: 'thin', color: { argb: 'FFCBD5E1' } },
      right:  { style: 'thin', color: { argb: 'FFCBD5E1' } },
    };
  });

  // ─── Данные ───────────────────────────────────────────────
  for (let i = 1; i <= count; i++) {
    const t = pick(TYPES);
    const model = pick(t.models);
    const vol = t.volumes.length ? pick(t.volumes) : '';

    // 30% — с явным ID (TEST-0001), 70% — без ID (проверка автогенерации)
    const useExplicitId = Math.random() < 0.30;
    const id = useExplicitId ? `TEST-${pad(i, 4)}` : '';

    const date = randomDate();
    const yearForCert = date.match(/(\d{4})/g)?.pop() || new Date().getFullYear();
    const cert = Math.random() < 0.85

    const notes = i % 25 === 0 ? 'Нагрузочный тест' : '';

    // Каждая 50-я — с невалидным типом (проверим fallback)
    let equipmentType = t.value;

    ws.addRow({
      id,
      serial: `SN-${pad(i, 6)}`,
      manufacturer: t.brand,
      model,
      equipmentType,
      volume: vol,
      department: pick(DEPARTMENTS),
      interval: pick([6, 12, 12, 12, 24]),
      lastCalibration: date,
      cert,
      result: pick(RESULTS),
      active: pick(ACTIVES),
      responsible: pick(RESPONSIBLES),
      location: pick(LOCATIONS),
      notes,
    });
  }

  // ─── Стилизация данных ────────────────────────────────────
  ws.eachRow((row, rowNumber) => {
    if (rowNumber === 1) return;

    row.height = 18;

    row.eachCell((cell, colNumber) => {
      cell.alignment = { vertical: 'middle', horizontal: 'left' };
      cell.border = {
        top:    { style: 'hair', color: { argb: 'FFE2E8F0' } },
        left:   { style: 'hair', color: { argb: 'FFE2E8F0' } },

        right:  { style: 'hair', color: { argb: 'FFE2E8F0' } },

      if ([1, 6, 8, 12].includes(colNumber)) {

        cell.font = { bold: true, color: { argb: 'FF991B1B' } };
        cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FFFEE2E2' } };

      }
    });
  });

  ws.autoFilter = {
    from: 'A1',
    to:   `O${count + 1}`,
  };

  // ─── Дополнительный лист: сводка ──────────────────────────
  const ws2 = wb.addWorksheet('Сводка');
  ws2.columns = [
    { header: 'Параметр', key: 'param', width: 30 },
    { header: 'Значение', key: 'value', width: 20 },

  ws2.addRow({ param: 'Без ID (автогенерация)',      value: Math.round(count * 0.70) });

  ws2.addRow({ param: 'Типов оборудования',          value: '6 (пипетка, анализатор, термометр, весы, фотометр, микроскоп)' });

  // ─── Сохранение ───────────────────────────────────────────
  await wb.xlsx.writeFile(filename);
  console.log(`✅ Создан файл: ${filename}`);
  console.log(`   Записей: ${count}`);
  console.log(`   Колонок: 15`);
  console.log(`   Формат: Excel (XLSX)`);
}

});

Что внутри
Лист «Оборудование»:

15 колонок с правильными заголовками (совпадают с HEADER_MAP в import.js)

Шапка тёмно-синяя, закреплена при скролле (freeze panes)

Автофильтр по всем колонкам

Зебра (чётные строки светло-серые)

«Брак» — красный жирный, «В процессе» — оранжевый, «Годен» — зелёный

«Нет» в статусе — серый

Смешивание форматов:

Даты: 30% ISO (2024-10-21), 35% русский (21.10.2024), 35% US (10.21.2024)

ID: 30% с явным ID (TEST-0001), 70% без ID → проверка автогенерации

Свидетельства: 85% заполнены, 15% пустые

Типы: случайно все 6 из справочника

Результаты: 8 «Годен» : 1 «Брак» : 1 «В процессе»

Активность: 4 «Да» : 1 «Нет»

Лист «Сводка» — маленькая табличка, какие данные внутри (для отчёта).

Как проверять
Открыть приложение под admin

«📥 Импорт» → выбрать test-import-1000.xlsx → «📥 Загрузить»

Дождаться тоста «Импортировано: N, пропущено: M»

Открыть F12 → Console:

res.skippedDetails — какие строки пропущены (только те, что с TEST-XXXX и совпали с существующими — этого быть не должно, если БД чистая)

res.errorDetails — если есть ошибки

Проверить в БД:

bash
docker exec -it pipette_db mysql -upipette -ppipette_secret pipette \
  -e "SELECT equipment_type, COUNT(*) FROM pipettes GROUP BY equipment_type;"
Должны быть все 6 типов

Проверить даты:

bash
docker exec -it pipette_db mysql -upipette -ppipette_secret pipette \
  -e "SELECT id, last_calibration FROM pipettes WHERE last_calibration NOT REGEXP '^[0-9]{4}-[0-9]{2}-[0-9]{2}$' LIMIT 5;"
Должно быть пусто — все даты нормализованы в ISO

Проверить историю:

bash
docker exec -it pipette_db mysql -upipette -ppipette_secret pipette \
  -e "SELECT COUNT(*) FROM calibration_history;"
Должно быть ≈ N (для строк с датой)

function addSheet(wb, name, headers, rows) {
  const ws = wb.addWorksheet(name, { views: [{ state: 'frozen', ySplit: 1 }] });
  ws.addRow(headers);

  const headerRow = ws.getRow(1);
  headerRow.height = 24;
  headerRow.eachCell(cell => {
    cell.font      = { bold: true, color: { argb: 'FFFFFFFF' }, size: 11 };
    cell.fill      = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF1E293B' } };
    cell.alignment = { vertical: 'middle', horizontal: 'left' };
    cell.border    = {
      top:{style:'thin',color:{argb:'FFCBD5E1'}}, left:{style:'thin',color:{argb:'FFCBD5E1'}},
      bottom:{style:'thin',color:{argb:'FFCBD5E1'}}, right:{style:'thin',color:{argb:'FFCBD5E1'}},
    };
  });

  for (const row of rows) ws.addRow(row);

  // Автоширина
  ws.columns.forEach((col, i) => {
    let max = 10;
    col.eachCell({ includeEmpty: true }, cell => {
      const len = cell.value ? String(cell.value).length : 0;
      if (len > max && len < 40) max = len;
    });
    col.width = max + 2;
  });

  ws.autoFilter = { from: 'A1', to: `${String.fromCharCode(64 + headers.length)}${rows.length + 1}` };
  return ws;
}

function cleanRow(i, opts = {}) {
  const t = pick(TYPES);
  const date = randomDate();
  const yearForCert = date.match(/(\d{4})/g)?.pop() || new Date().getFullYear();
  return [
    opts.useId ? `TEST-${pad(i, 4)}` : '',

    t.brand,
    pick(t.models),
    t.value,
    t.volumes.length ? pick(t.volumes) : '',
    pick(DEPARTMENTS),
    pick([6, 12, 12, 12, 24]),
    date,
    Math.random() < 0.85 ? `С-АБ-${pad(i, 6)}/${yearForCert}` : '',
    pick(RESULTS),
    pick(ACTIVES),
    pick(RESPONSIBLES),
    pick(LOCATIONS),
    i % 25 === 0 ? 'Нагрузочный тест' : '',
  ];
}

// Возвращает [row, comment] — где comment описывает, что сломано
function brokenRow(i) {
  const seed = i % 16;
  const base = () => [
    '', `SN-${pad(i, 6)}`, 'TestBrand', 'TestModel', 'Пипетка', '100',
    'Гематологический отдел', 12, '2024-01-01', 'С-АБ-0001/2024',
    'Годен', 'Да', 'Тестов Т.Т.', 'Лаб. 101', ''
  ];
  const row = base();
  const r = Math.random;

  switch (seed) {
    case 0: // Дубли ID (проверим на 3-х строках подряд)
      row[0] = 'DUP-001'; return [row, 'DUP-ID: DUP-001'];
    case 1:
      row[0] = 'DUP-001'; return [row, 'DUP-ID повтор'];
    case 2:
      row[0] = 'DUP-001'; return [row, 'DUP-ID ещё раз'];

    case 3: // Пустая модель
      row[3] = ''; return [row, 'Пустая модель'];

    case 4: // Битые даты
      row[8] = '32.13.2024'; return [row, 'Невалидный день/месяц'];

    case 5:
      row[8] = 'не дата'; return [row, 'Дата текстом'];

    case 6:
      row[8] = '2024-13-45'; return [row, 'ISO с невалидным месяцем'];

    case 7:
      row[8] = '00.00.0000'; return [row, 'Нулевая дата'];

    case 8: // XSS
      row[3] = '<script>alert(1)</script>';
      return [row, 'XSS в модели'];

    case 9: // SQL-инъекция в ID
      row[0] = "P-001'; DROP TABLE pipettes; --";
      return [row, 'SQL-инъекция в ID'];

    case 10: // Очень длинный текст
      row[14] = 'A'.repeat(3000);
      return [row, 'Очень длинный текст (3000 символов)'];

    case 11: // Дата в будущем
      row[8] = '2099-12-31';
      return [row, 'Дата в будущем'];

    case 12: // Невалидный интервал
      row[7] = 'abc';
      return [row, 'Интервал текстом'];

    case 13: // Отрицательный интервал
      row[7] = -12;
      return [row, 'Отрицательный интервал'];

    case 14: // Неизвестный тип оборудования
      row[4] = 'Космический корабль';
      return [row, 'Неизвестный тип'];

    case 15: // Результат вне допустимых
      row[10] = 'Абракадабра';
      return [row, 'Невалидный результат'];
  }
  return [base(), ''];
}

// ─── Файл 1: чистый (1000 записей) ───────────────────────────
async function makeClean() {
  const wb = new ExcelJS.Workbook();
  const rows = [];
  for (let i = 1; i <= 1000; i++) rows.push(cleanRow(i, { useId: i % 3 === 0 }));
  addSheet(wb, 'Оборудование', DEFAULT_HEADERS, rows);
  await wb.xlsx.writeFile('test-import-clean-1000.xlsx');

// ─── Файл 2: смешанный (700 чистых + 300 ошибок) ────────────
async function makeMixed() {
  const wb = new ExcelJS.Workbook();
  const rows = [];

  // Сначала 300 «плохих» (первая половина — с ошибками)
  for (let i = 1; i <= 300; i++) {
    const [row] = brokenRow(i);
    rows.push(row);
  }
  // Потом 700 «чистых»
  for (let i = 301; i <= 1000; i++) {
    rows.push(cleanRow(i, { useId: false }));
  }

  // Перемешать
  for (let i = rows.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [rows[i], rows[j]] = [rows[j], rows[i]];
  }

  addSheet(wb, 'Оборудование', DEFAULT_HEADERS, rows);
  await wb.xlsx.writeFile('test-import-mixed-1000.xlsx');
  console.log('✅ test-import-mixed-1000.xlsx (300 ошибок + 700 чистых)');
}

// ─── Файл 3: только ошибки (30 строк) ────────────────────────
async function makeErrorsOnly() {

// ─── Файл 5: с формулами Excel ───────────────────────────────
async function makeFormulas() {
  const wb = new ExcelJS.Workbook();
  const ws = wb.addWorksheet('Формулы');
  ws.addRow(DEFAULT_HEADERS);

  const headerRow = ws.getRow(1);
  headerRow.eachCell(cell => {
    cell.font = { bold: true, color: { argb: 'FFFFFFFF' } };
    cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF1E293B' } };
  });

  for (let i = 1; i <= 20; i++) {
    const row = cleanRow(i, { useId: false });
    const wsRow = ws.addRow(row);
    // Меняем некоторые ячейки на формулы
    wsRow.getCell(1).value = { formula: `"FORMULA-${pad(i,3)}"`, result: `FORMULA-${pad(i,3)}` };
    wsRow.getCell(4).value = { formula: `CONCATENATE("Model-", ${i})`, result: `Model-${i}` };
    wsRow.getCell(15).value = { formula: `"Сумма: " & ${i*10}`, result: `Сумма: ${i*10}` };
  }

  await wb.xlsx.writeFile('test-import-formulas.xlsx');
  console.log('✅ test-import-formulas.xlsx (20 строк с формулами)');
}

Проверки в UI
F12 → Console — после импорта смотри skippedDetails и errorDetails (там список пропущенных/ошибочных).

Тост — формат: Импортировано: N, пропущено: M, ошибок: K.

Таблица — фильтр по «Модель» покажет XSS-строку как текст, а не как скрипт.

История — у XSS-строки в поле «Модель» отображается <script>alert(1)</script> как текст, всплытия нет.
