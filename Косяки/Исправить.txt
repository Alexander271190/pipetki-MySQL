2. updateSortArrows() не вызывается при пустом фильтре
В render():

js
if (filtered.length === 0) {
  tbody.innerHTML = '';
  table.style.display = 'none';
  empty.style.display = 'block';
  if (pipettes.length > 0) empty.querySelector('p').textContent = 'Ничего не найдено по фильтру.';
  updateSelectAllCheckbox();
  updateBulkCalButton();
  return;                          // ← стрелки сортировки не обновляются
}
Если пользователь отфильтровал всё до нуля, потом поменял сортировку — стрелки в шапке останутся старыми. Добавь updateSortArrows(); перед return;.

3. Бэкенд: only_own_department не проверяется на GET /:id и GET /:id/calibration
В routes/pipettes.js:

js
router.get('/:id', authenticate, async (req, res) => {
  const [rows] = await db.query('SELECT * FROM pipettes WHERE id = ?', [req.params.id]);
  ...
});
и

js
router.get('/:id/calibration', authenticate, async (req, res) => {
  const [rows] = await db.query(
    'SELECT * FROM calibration_history WHERE pipette_id = ? ...'
  );
  ...
});
Пользователь с only_own_department = 1 в списке видит только свой отдел, но по прямому ID (или через историю) может открыть чужое оборудование. Список фильтруется, деталь — нет.

Если это не критично для лаборатории — можно оставить. Если критично — добавить проверку:

js
if (req.user.role !== 'admin' && req.user.only_own_department && req.user.department
    && rows[0].department !== req.user.department) {
  return res.status(403).json({ error: 'Нет доступа к этому оборудованию' });
}
