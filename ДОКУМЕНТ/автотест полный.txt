   * ═══════════════════════════════════════════════════════════ */

      actSendFields: ['id', 'model', 'serial'],
      actReturnFields: ['id', 'cert', 'result'],

    // Обновим settings.printDocs через прямой запрос
    const eff = await api('/settings/print-documents/effective');
    if (eff.ok) settings.printDocs = eff.body;

      if (fakeRSingle) fakeRSingle.checked = false;

      _openCallCount = 0; _lastOpenHtml = '';
      try { await printReturnAct(); await sleep(200); } catch (e) { log('PRINT', 'printReturnAct вызов', 'FAIL', e.message); }
      log('PRINT', 'printReturnAct — window.open вызван', _openCallCount >= 1 ? 'PASS' : 'FAIL', `${_openCallCount}`);

   * ═══════════════════════════════════════════════════════════ */

      log('PRINT', 'PDF history — колонка «Дата»', /Дата/.test(pdfHtml) ? 'PASS' : 'FAIL');
      log('PRINT', 'PDF history — колонка «Свидетельство»', /Свидетельство/.test(pdfHtml) ? 'PASS' : 'FAIL');
      log('PRINT', 'PDF history — НЕТ колонки «Организация»', !/Организация<\/th>/.test(pdfHtml) ? 'PASS' : 'FAIL');
    } else {
      log('PRINT', 'historyFields', 'SKIP', 'у первого прибора нет истории');
    }

    await api('/settings/export', 'PUT', _origExportCfg);
    const effR = await api('/settings/print-documents/effective');
    if (effR.ok) settings.printDocs = effR.body;
