      await api('/pipettes/bulk-send', 'POST', { ids: [freshId], sentDate: today, note: '' });
      await sleep(300);
      await reloadPipettes();  // ← ОБЯЗАТЕЛЬНО перед return
      await sleep(200);

      if (typeof openBulkReturnModal === 'function') {
        try { await openBulkReturnModal(); await sleep(300); } catch (e) { log('PRINT', 'openBulkReturnModal', 'FAIL', e.message); }

      if (typeof printReturnAct === 'function') {
        _bulkReturnIds = [freshId];
        const fakeRDate = document.getElementById('bulk-return-date');
        if (fakeRDate) fakeRDate.value = today;

        try { await printReturnAct(); await sleep(300); }

            `open=${_openCallCount}, html=${retHtml.length}b`);

            /Внутр/.test(retHtml) && /Свидетельство/.test(retHtml) && /Результат/.test(retHtml) ? 'PASS' : 'FAIL',

  {
    const getTheme = () => {
      const h = document.documentElement, b = document.body;
      const cls = (h.className + ' ' + b.className).toLowerCase();
      if (cls.includes('dark')) return 'dark';
      if (cls.includes('light')) return 'light';
      return h.getAttribute('data-theme') || b.getAttribute('data-theme') || localStorage.getItem('theme') || 'unknown';
    };
    const before = getTheme();
    let toggled = false;
    if (typeof window.toggleTheme === 'function') {
      try { window.toggleTheme(); toggled = true; } catch (e) {}
    } else if (typeof window.setTheme === 'function') {
      try { window.setTheme(before === 'dark' ? 'light' : 'dark'); toggled = true; } catch (e) {}
    } else {
      const btn = document.querySelector('[onclick*="oggleTheme"], [onclick*="etTheme"], [onclick*="witchTheme"]');
      if (btn) { btn.click(); toggled = true; }
    }
    if (toggled) {
      await sleep(400);
      const after = getTheme();
      log('THEME', 'переключение', before !== after ? 'PASS' : 'WARN', `${before} → ${after}`);
      if (typeof window.toggleTheme === 'function') { try { window.toggleTheme(); } catch(e){} }
      else if (typeof window.setTheme === 'function') { try { window.setTheme(before); } catch(e){} }
      else {
        const btn = document.querySelector('[onclick*="oggleTheme"], [onclick*="etTheme"], [onclick*="witchTheme"]');
